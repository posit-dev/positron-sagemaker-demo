# The packages needed to render, and no more. Posit Connect rebuilds this list.
#
# This list is shorter than pyproject.toml. The reports call
# boto3.client("sagemaker-runtime").invoke_endpoint(...) and never import the
# sagemaker SDK, which installs torch and mlflow. Those two are the largest
# cause of a failed install on Connect.
#
# The reports read Athena over ODBC, so they need pyodbc and an Athena driver
# on the Connect server. They no longer need awswrangler.
boto3
pyodbc
pandas
pyarrow
matplotlib
great-tables
jupyter
nbformat
nbclient
